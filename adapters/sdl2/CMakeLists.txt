cmake_minimum_required(VERSION 3.20)
project(rengine_sdl_surface LANGUAGES CXX)
find_package(SDL2 2.32.10 EXACT CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
add_library(rengine_surface SHARED surface.cpp)
target_compile_definitions(rengine_surface PRIVATE RENGINE_SURFACE_BUILD)
target_compile_features(rengine_surface PRIVATE cxx_std_17)
target_link_libraries(rengine_surface PRIVATE SDL2::SDL2 OpenGL::GL Threads::Threads)
if(APPLE)
  target_sources(rengine_surface PRIVATE interpose_macos.cpp)
  target_compile_options(rengine_surface PRIVATE -Wno-deprecated-declarations)
elseif(WIN32)
  target_link_libraries(rengine_surface PRIVATE ws2_32)
endif()
add_executable(rengine_surface_fixture ../../orchestrator/tests/surface_fixture.cpp)
target_compile_features(rengine_surface_fixture PRIVATE cxx_std_17)
target_link_libraries(rengine_surface_fixture PRIVATE SDL2::SDL2 OpenGL::GL)
if(APPLE)
  target_compile_options(rengine_surface_fixture PRIVATE -Wno-deprecated-declarations)
elseif(WIN32)
  target_include_directories(rengine_surface_fixture PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(rengine_surface_fixture PRIVATE rengine_surface SDL2::SDL2main)
endif()

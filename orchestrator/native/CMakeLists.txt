find_package(SDL2 2.32.10 EXACT CONFIG REQUIRED)
include(curl.cmake)
set(RENGINE_VENDOR "${PROJECT_SOURCE_DIR}/third_party")

add_library(rengine_microui STATIC "${RENGINE_VENDOR}/microui/microui.c")
target_include_directories(rengine_microui SYSTEM PUBLIC "${RENGINE_VENDOR}/microui")
add_library(rengine_json STATIC "${RENGINE_VENDOR}/cjson/cJSON.c")
target_include_directories(rengine_json SYSTEM PUBLIC "${RENGINE_VENDOR}/cjson")
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS "${RENGINE_VENDOR}/libvterm/src/*.c")
add_library(rengine_vterm STATIC ${VTERM_SOURCES})
target_include_directories(rengine_vterm SYSTEM PUBLIC "${RENGINE_VENDOR}/libvterm/include")
target_compile_definitions(rengine_vterm PRIVATE _CRT_SECURE_NO_WARNINGS)

add_library(rengine_desktop_core STATIC draw.c layout.c net.c terminal.c editor.c app.c workspace.c game.c scroll.c
  render/utf8.c render/draw_list.c render/font.c render/backend_sdl.c render/backend_gl.c scene.c)
target_include_directories(rengine_desktop_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(rengine_desktop_core SYSTEM PRIVATE "${RENGINE_VENDOR}/stb" "${RENGINE_VENDOR}")
target_link_libraries(rengine_desktop_core PUBLIC SDL2::SDL2 rengine_microui rengine_json rengine_vterm CURL::libcurl)
target_compile_features(rengine_desktop_core PUBLIC c_std_11)
target_compile_definitions(rengine_desktop_core PUBLIC SDL_MAIN_HANDLED)
if(NOT MSVC)
  target_compile_options(rengine_desktop_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(rengine_desktop_core PUBLIC m)
endif()
find_program(RENGINE_NODE_EXECUTABLE NAMES node REQUIRED)
file(TO_CMAKE_PATH "${RENGINE_NODE_EXECUTABLE}" RENGINE_NODE_PATH)
add_executable(rengine main.c automation.c bootstrap.c)
target_compile_definitions(rengine PRIVATE RENGINE_NODE="${RENGINE_NODE_PATH}"
  RENGINE_BOOTSTRAP="${PROJECT_SOURCE_DIR}/orchestrator/runtime/bootstrap.mjs")
target_link_libraries(rengine PRIVATE rengine_desktop_core)
set_target_properties(rengine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  add_custom_command(TARGET rengine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:rengine>)
endif()

  if(MSVC)

  else()
    target_compile_options(rengine_layout_test PRIVATE -UNDEBUG)

  add_test(NAME native_layout COMMAND rengine_layout_test)

  add_test(NAME native_editor COMMAND rengine_editor_test)

  if(MSVC)
    target_compile_options(rengine_terminal_test PRIVATE /UNDEBUG)

    target_compile_options(rengine_terminal_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME native_terminal COMMAND rengine_terminal_test)
